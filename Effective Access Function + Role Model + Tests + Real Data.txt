/**
 * Function: rbac.evaluate_access
 * 
 * PRIMARY AUTHORIZATION DECISION FUNCTION
 * Single source of truth for ALLOW/DENY decisions
 * 
 * Parameters:
 *   p_user_id        - User ID requesting access
 *   p_menu_code      - Menu code (e.g., 'NURSE_MASTER')
 *   p_permission_code- Permission code (e.g., 'EDIT')
 *   p_resource_id    - Optional: specific record ID (for data scope validation)
 * 
 * Returns:
 *   decision         - 'ALLOW' or 'DENY'
 *   reason           - Explanation of decision
 *   menu_accessible  - Is menu visible AND enabled?
 *   permission_granted - Is permission allowed?
 *   data_scope_valid - Is resource within scope?
 *   cache_ttl        - Recommended cache time (seconds)
 * 
 * AND-Logic:
 *   ALLOW only if ALL of:
 *     1. User is Active
 *     2. Menu is Active and accessible
 *     3. Permission is Active and granted
 *     4. (If resource_id) Resource is within data scope
 *     5. All temporal checks pass (effective_from/to)
 * 
 * Deny by Default:
 *   Missing record = DENY (no exceptions)
 */
CREATE OR REPLACE FUNCTION rbac.evaluate_access(
  p_user_id BIGINT,
  p_menu_code VARCHAR(100),
  p_permission_code VARCHAR(50),
  p_resource_id BIGINT DEFAULT NULL
)
RETURNS TABLE (
  decision VARCHAR(50),           -- 'ALLOW' or 'DENY'
  reason VARCHAR(500),            -- Detailed explanation
  menu_accessible BOOLEAN,        -- Menu visible AND enabled?
  permission_granted BOOLEAN,     -- Permission allowed?
  data_scope_valid BOOLEAN,       -- Resource in scope?
  cache_ttl INT,                  -- Recommended cache TTL (seconds)
  evaluated_at TIMESTAMP,         -- When evaluated
  user_role VARCHAR(50)           -- User's primary role (for audit)
) AS $$
DECLARE
  -- User validation
  v_user_status VARCHAR(50);
  v_user_primary_role_id BIGINT;
  v_user_primary_role_code VARCHAR(50);
  v_now TIMESTAMP := CURRENT_TIMESTAMP;
  
  -- Menu validation
  v_menu_id BIGINT;
  v_menu_status VARCHAR(50);
  
  -- Role-Menu Access validation
  v_menu_visible BOOLEAN := FALSE;
  v_menu_enabled BOOLEAN := FALSE;
  v_menu_effective_from TIMESTAMP;
  v_menu_effective_to TIMESTAMP;
  v_menu_found BOOLEAN := FALSE;
  
  -- Role-Permission validation
  v_perm_id BIGINT;
  v_perm_status VARCHAR(50);
  v_perm_allowed BOOLEAN := FALSE;
  v_perm_effective_from TIMESTAMP;
  v_perm_effective_to TIMESTAMP;
  v_perm_found BOOLEAN := FALSE;
  
  -- Data scope validation
  v_scope_valid BOOLEAN := FALSE;
  v_scope_count INT := 0;
  
  -- Decision tracking
  v_decision VARCHAR(50) := 'DENY';
  v_reason VARCHAR(500) := 'Authorization check failed';
  v_cache_ttl INT := 300; -- Default 5 min
  
BEGIN
  -- ========================================================================
  -- STEP 1: VALIDATE USER AND GET PRIMARY ROLE
  -- ========================================================================
  
  SELECT u.status, u.primary_role_id, hr.code
  INTO v_user_status, v_user_primary_role_id, v_user_primary_role_code
  FROM auth.users u
  LEFT JOIN system.hospital_roles hr ON hr.id = u.primary_role_id
  WHERE u.id = p_user_id;
  
  -- User does not exist
  IF v_user_status IS NULL THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'User not found'::VARCHAR(500),
      FALSE, FALSE, FALSE, 300, v_now, NULL::VARCHAR(50);
    RETURN;
  END IF;
  
  -- User is not Active
  IF v_user_status != 'Active' THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'User status is ' || v_user_status || ', not Active'::VARCHAR(500),
      FALSE, FALSE, FALSE, 300, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- User has no primary role
  IF v_user_primary_role_code IS NULL THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'User has no assigned primary role'::VARCHAR(500),
      FALSE, FALSE, FALSE, 300, v_now, NULL::VARCHAR(50);
    RETURN;
  END IF;
  
  -- ========================================================================
  -- STEP 2: VALIDATE MENU
  -- ========================================================================
  
  SELECT id, status
  INTO v_menu_id, v_menu_status
  FROM rbac.menus
  WHERE code = p_menu_code;
  
  -- Menu does not exist
  IF v_menu_id IS NULL THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'Menu "' || p_menu_code || '" not found'::VARCHAR(500),
      FALSE, FALSE, FALSE, 300, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- Menu is not Active
  IF v_menu_status != 'Active' THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'Menu is inactive'::VARCHAR(500),
      FALSE, FALSE, FALSE, 300, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- ========================================================================
  -- STEP 3: CHECK ROLE-MENU ACCESS (Visibility & Enablement)
  -- ========================================================================
  
  SELECT 
    rma.visible,
    rma.enabled,
    rma.effective_from,
    rma.effective_to
  INTO 
    v_menu_visible,
    v_menu_enabled,
    v_menu_effective_from,
    v_menu_effective_to
  FROM rbac.role_menu_access rma
  WHERE rma.role_code = v_user_primary_role_code
    AND rma.menu_id = v_menu_id
    AND rma.status = 'Active';
  
  -- Record found, validate temporal bounds
  IF FOUND THEN
    v_menu_found := TRUE;
    
    -- Check effective_from
    IF v_menu_effective_from IS NOT NULL AND v_now < v_menu_effective_from THEN
      v_menu_visible := FALSE;
      v_menu_enabled := FALSE;
    END IF;
    
    -- Check effective_to
    IF v_menu_effective_to IS NOT NULL AND v_now > v_menu_effective_to THEN
      v_menu_visible := FALSE;
      v_menu_enabled := FALSE;
    END IF;
  ELSE
    -- No role_menu_access record = deny by default
    v_menu_found := FALSE;
    v_menu_visible := FALSE;
    v_menu_enabled := FALSE;
  END IF;
  
  -- Menu access check FAILED
  IF NOT (v_menu_visible AND v_menu_enabled) THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'Menu not accessible (visible=' || v_menu_visible::TEXT || 
        ', enabled=' || v_menu_enabled::TEXT || ')'::VARCHAR(500),
      v_menu_visible, FALSE, FALSE, 1800, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- ========================================================================
  -- STEP 4: VALIDATE PERMISSION EXISTS
  -- ========================================================================
  
  SELECT id, status
  INTO v_perm_id, v_perm_status
  FROM rbac.permissions
  WHERE code = p_permission_code;
  
  -- Permission does not exist
  IF v_perm_id IS NULL THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'Permission "' || p_permission_code || '" not found'::VARCHAR(500),
      TRUE, FALSE, FALSE, 1800, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- Permission is not Active
  IF v_perm_status != 'Active' THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'Permission is inactive'::VARCHAR(500),
      TRUE, FALSE, FALSE, 1800, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- ========================================================================
  -- STEP 5: CHECK ROLE-PERMISSION (Permission Grant)
  -- ========================================================================
  
  SELECT 
    rp.allowed,
    rp.effective_from,
    rp.effective_to
  INTO 
    v_perm_allowed,
    v_perm_effective_from,
    v_perm_effective_to
  FROM rbac.role_permissions rp
  WHERE rp.role_code = v_user_primary_role_code
    AND rp.menu_id = v_menu_id
    AND rp.permission_id = v_perm_id
    AND rp.status = 'Active';
  
  -- Record found, validate temporal bounds
  IF FOUND THEN
    v_perm_found := TRUE;
    
    -- Check effective_from
    IF v_perm_effective_from IS NOT NULL AND v_now < v_perm_effective_from THEN
      v_perm_allowed := FALSE;
    END IF;
    
    -- Check effective_to
    IF v_perm_effective_to IS NOT NULL AND v_now > v_perm_effective_to THEN
      v_perm_allowed := FALSE;
    END IF;
  ELSE
    -- No role_permission record = deny by default
    v_perm_found := FALSE;
    v_perm_allowed := FALSE;
  END IF;
  
  -- Permission check FAILED
  IF NOT v_perm_allowed THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'Permission "' || p_permission_code || '" not granted'::VARCHAR(500),
      TRUE, FALSE, FALSE, 1800, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- ========================================================================
  -- STEP 6: CHECK DATA SCOPE (if resource_id provided)
  -- ========================================================================
  
  v_scope_valid := TRUE; -- Default: no resource ID = scope doesn't apply
  
  IF p_resource_id IS NOT NULL THEN
    -- Count valid data scopes for user
    SELECT COUNT(*)
    INTO v_scope_count
    FROM rbac.user_data_scopes uds
    WHERE uds.user_id = p_user_id
      AND uds.status = 'Active'
      AND (uds.effective_from IS NULL OR uds.effective_from <= v_now)
      AND (uds.effective_to IS NULL OR uds.effective_to >= v_now);
    
    -- User must have at least one valid scope to access resources
    IF v_scope_count = 0 THEN
      v_scope_valid := FALSE;
    ELSE
      -- TODO: Implement resource-specific scope validation
      -- This requires knowing what menu/resource type we're accessing
      -- For now: if user has ANY scope, resource access is valid
      v_scope_valid := TRUE;
    END IF;
  END IF;
  
  -- Data scope check FAILED
  IF NOT v_scope_valid THEN
    RETURN QUERY SELECT 
      'DENY'::VARCHAR(50),
      'Resource outside user data scope'::VARCHAR(500),
      TRUE, TRUE, FALSE, 1800, v_now, v_user_primary_role_code;
    RETURN;
  END IF;
  
  -- ========================================================================
  -- STEP 7: ALL CHECKS PASSED - ALLOW
  -- ========================================================================
  
  RETURN QUERY SELECT 
    'ALLOW'::VARCHAR(50),
    'Authorization granted: all checks passed'::VARCHAR(500),
    v_menu_visible,
    v_perm_allowed,
    v_scope_valid,
    300, -- Cache allows for 5 min (temporal filtering may expire)
    v_now,
    v_user_primary_role_code;
  
END;
$$ LANGUAGE plpgsql STABLE;

/**
 * Function: rbac.get_user_full_access
 * 
 * Returns complete access matrix for a user
 * Shows all menus, permissions, and accessible status
 * Used for:
 *   - Building user dashboard
 *   - Audit reports
 *   - Access verification
 */
CREATE OR REPLACE FUNCTION rbac.get_user_full_access(p_user_id BIGINT)
RETURNS TABLE (
  user_id BIGINT,
  username VARCHAR(100),
  primary_role_code VARCHAR(50),
  primary_role_name VARCHAR(100),
  menu_id BIGINT,
  menu_code VARCHAR(100),
  menu_name VARCHAR(255),
  menu_route VARCHAR(500),
  permission_id BIGINT,
  permission_code VARCHAR(50),
  permission_name VARCHAR(255),
  is_accessible BOOLEAN,  -- Menu visible AND enabled?
  is_allowed BOOLEAN      -- Permission allowed?
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    u.id,
    u.username,
    hr.code,
    hr.name,
    m.id,
    m.code,
    m.name,
    m.route,
    p.id,
    p.code,
    p.name,
    -- Menu accessible?
    COALESCE(rma.visible AND rma.enabled, FALSE),
    -- Permission allowed?
    COALESCE(rp.allowed, FALSE)
  FROM auth.users u
  LEFT JOIN system.hospital_roles hr ON hr.id = u.primary_role_id
  CROSS JOIN rbac.menus m
  CROSS JOIN rbac.permissions p
  LEFT JOIN rbac.role_menu_access rma 
    ON rma.role_code = hr.code 
    AND rma.menu_id = m.id 
    AND rma.status = 'Active'
    AND (rma.effective_from IS NULL OR rma.effective_from <= CURRENT_TIMESTAMP)
    AND (rma.effective_to IS NULL OR rma.effective_to >= CURRENT_TIMESTAMP)
  LEFT JOIN rbac.role_permissions rp 
    ON rp.role_code = hr.code 
    AND rp.menu_id = m.id 
    AND rp.permission_id = p.id 
    AND rp.status = 'Active'
    AND (rp.effective_from IS NULL OR rp.effective_from <= CURRENT_TIMESTAMP)
    AND (rp.effective_to IS NULL OR rp.effective_to >= CURRENT_TIMESTAMP)
  WHERE u.id = p_user_id
    AND u.status = 'Active'
    AND m.status = 'Active'
    AND p.status = 'Active';
END;
$$ LANGUAGE plpgsql STABLE;

/**
 * Function: rbac.preview_access_change
 * 
 * Shows what access would be if a configuration change were made
 * WITHOUT actually committing the change
 * 
 * Used for impact analysis before making security-sensitive changes
 */
CREATE OR REPLACE FUNCTION rbac.preview_access_change(
  p_user_id BIGINT,
  p_change_type VARCHAR(50), -- 'PERMISSION_GRANT', 'MENU_ACCESS_GRANT', 'SCOPE_CHANGE'
  p_menu_id BIGINT,
  p_permission_id BIGINT DEFAULT NULL,
  p_new_value BOOLEAN DEFAULT NULL
)
RETURNS TABLE (
  current_decision VARCHAR(50),
  proposed_decision VARCHAR(50),
  menu_code VARCHAR(100),
  menu_name VARCHAR(255),
  permission_code VARCHAR(50),
  permission_name VARCHAR(255),
  impact_description VARCHAR(500)
) AS $$
DECLARE
  v_menu_code VARCHAR(100);
  v_menu_name VARCHAR(255);
  v_perm_code VARCHAR(50);
  v_perm_name VARCHAR(255);
  v_current_decision VARCHAR(50);
  v_proposed_decision VARCHAR(50);
  v_role_code VARCHAR(50);
BEGIN
  -- Get menu and permission codes
  SELECT code, name INTO v_menu_code, v_menu_name
  FROM rbac.menus WHERE id = p_menu_id;
  
  IF p_permission_id IS NOT NULL THEN
    SELECT code, name INTO v_perm_code, v_perm_name
    FROM rbac.permissions WHERE id = p_permission_id;
  END IF;
  
  -- Get user's role
  SELECT hr.code INTO v_role_code
  FROM auth.users u
  LEFT JOIN system.hospital_roles hr ON hr.id = u.primary_role_id
  WHERE u.id = p_user_id;
  
  -- Get current decision
  SELECT decision INTO v_current_decision
  FROM rbac.evaluate_access(p_user_id, v_menu_code, v_perm_code);
  
  -- TODO: Calculate proposed decision based on change_type
  -- For now, simulate
  v_proposed_decision := v_current_decision;
  
  RETURN QUERY SELECT 
    v_current_decision,
    v_proposed_decision,
    v_menu_code,
    v_menu_name,
    v_perm_code,
    v_perm_name,
    'Impact analysis not yet implemented'::VARCHAR(500);
END;
$$ LANGUAGE plpgsql STABLE;

/**
 * Table: system.hospital_roles
 * 
 * Master list of all hospital roles
 * Examples:
 *   - RN (Registered Nurse)
 *   - Charge Nurse
 *   - Nurse Manager
 *   - Scheduler
 *   - HR Administrator
 *   - System Administrator
 * 
 * NOT temporary, NOT for casual access requests
 * These are REAL organizational roles with legal/compliance implications
 */
CREATE TABLE IF NOT EXISTS system.hospital_roles (
  id BIGSERIAL PRIMARY KEY,
  
  -- Unique code (e.g., 'RN', 'CHARGE_NURSE')
  code VARCHAR(50) UNIQUE NOT NULL,
  
  -- Display name (e.g., 'Registered Nurse')
  name VARCHAR(100) NOT NULL UNIQUE,
  
  -- Detailed description of role purpose and responsibilities
  description TEXT NOT NULL,
  
  -- Category for grouping
  category VARCHAR(50) NOT NULL 
    CHECK (category IN ('Clinical', 'Administrative', 'System', 'Support')),
  
  -- Which department this role belongs to
  department VARCHAR(100),
  
  -- Which roles can assign this role
  assignable_by_roles VARCHAR(50)[] DEFAULT ARRAY['SYSTEM_ADMIN'],
  
  -- Can a user have multiple instances of this role? (e.g., across departments)
  is_exclusive BOOLEAN DEFAULT TRUE,
  
  -- Status
  status VARCHAR(50) DEFAULT 'Active' 
    CHECK (status IN ('Active', 'Inactive', 'Deprecated')),
  
  -- Audit
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  created_by BIGINT,
  updated_by BIGINT
);

COMMENT ON TABLE system.hospital_roles IS 'Master list of hospital organizational roles';
COMMENT ON COLUMN system.hospital_roles.code IS 'Unique identifier (e.g., RN, CHARGE_NURSE, NURSE_MANAGER)';
COMMENT ON COLUMN system.hospital_roles.category IS 'Role category for grouping and filtering';
COMMENT ON COLUMN system.hospital_roles.is_exclusive IS 'Can user have multiple of this role? Usually TRUE';

CREATE INDEX idx_hospital_roles_code ON system.hospital_roles(code);
CREATE INDEX idx_hospital_roles_category ON system.hospital_roles(category);
CREATE INDEX idx_hospital_roles_status ON system.hospital_roles(status);

/**
 * Table: auth.user_role_assignments
 * 
 * Many-to-many relationship between users and roles
 * A user can have multiple roles
 * A role can be assigned to multiple users
 * 
 * Supports:
 *   - Temporal role assignments (effective_from/to for temporary duties)
 *   - Role change tracking (who assigned, when, why)
 *   - Audit trail of role changes
 */
CREATE TABLE IF NOT EXISTS auth.user_role_assignments (
  id BIGSERIAL PRIMARY KEY,
  
  -- User being assigned
  user_id BIGINT NOT NULL,
  
  -- Role being assigned
  role_id BIGINT NOT NULL,
  
  -- Who assigned this role?
  assigned_by BIGINT NOT NULL,
  
  -- When does this role become effective?
  effective_from TIMESTAMP,
  
  -- When does this role expire?
  effective_to TIMESTAMP,
  
  -- Why was this role assigned? (e.g., "Covering for sick leave", "Promotion")
  reason TEXT,
  
  -- Status (to allow soft delete)
  status VARCHAR(50) DEFAULT 'Active' 
    CHECK (status IN ('Active', 'Inactive', 'Pending', 'Rejected')),
  
  -- Audit
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  -- Constraints
  CONSTRAINT fk_user_role_assignments_user 
    FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE,
  CONSTRAINT fk_user_role_assignments_role 
    FOREIGN KEY (role_id) REFERENCES system.hospital_roles(id) ON DELETE RESTRICT,
  CONSTRAINT fk_user_role_assignments_assigned_by 
    FOREIGN KEY (assigned_by) REFERENCES auth.users(id) ON DELETE SET NULL,
  
  -- Unique: same user cannot have same role twice with overlapping dates
  CONSTRAINT uk_user_role_assignments 
    UNIQUE (user_id, role_id),
  
  -- Business rule: effective_to must be after effective_from
  CONSTRAINT chk_user_role_effective_dates 
    CHECK (effective_to IS NULL OR effective_from IS NULL OR effective_from <= effective_to)
);

COMMENT ON TABLE auth.user_role_assignments IS 'Maps users to hospital roles (many-to-many)';
COMMENT ON COLUMN auth.user_role_assignments.effective_from IS 'When role becomes active (NULL = immediately)';
COMMENT ON COLUMN auth.user_role_assignments.effective_to IS 'When role expires (NULL = indefinitely)';
COMMENT ON COLUMN auth.user_role_assignments.reason IS 'Business reason for assignment (required for audit)';

CREATE INDEX idx_user_role_assignments_user 
  ON auth.user_role_assignments(user_id, status);
CREATE INDEX idx_user_role_assignments_role 
  ON auth.user_role_assignments(role_id, status);
CREATE INDEX idx_user_role_assignments_effective 
  ON auth.user_role_assignments(effective_from, effective_to);
CREATE INDEX idx_user_role_assignments_active 
  ON auth.user_role_assignments(user_id, status, effective_from, effective_to);

  ura.effective_from,

  (u.primary_role_id = hr.id) as is_primary,
  -- Is this role currently active (temporal bounds respected)?
  (ura.status = 'Active' 
    AND (ura.effective_from IS NULL OR ura.effective_from <= CURRENT_TIMESTAMP)
    AND (ura.effective_to IS NULL OR ura.effective_to >= CURRENT_TIMESTAMP)) as is_currently_active,
  u.status as user_status,
  hr.status as role_status
FROM auth.users u
LEFT JOIN auth.user_role_assignments ura ON ura.user_id = u.id
LEFT JOIN system.hospital_roles hr ON hr.id = ura.role_id
WHERE u.status = 'Active';

COMMENT ON VIEW auth.vw_user_current_roles IS 
  'Shows all currently active roles for users, respecting temporal bounds';

/**
 * Function: auth.get_user_primary_role
 * Returns the primary role code for a user
 * Used for backward compatibility with single-role references
 */
CREATE OR REPLACE FUNCTION auth.get_user_primary_role(p_user_id BIGINT)
RETURNS VARCHAR(50) AS $$
DECLARE
  v_role_code VARCHAR(50);
BEGIN
  SELECT hr.code INTO v_role_code
  FROM auth.users u
  LEFT JOIN system.hospital_roles hr ON hr.id = u.primary_role_id
  WHERE u.id = p_user_id AND u.status = 'Active'
  LIMIT 1;
  
  RETURN COALESCE(v_role_code, 'READONLY_USER');
END;
$$ LANGUAGE plpgsql STABLE;

COMMENT ON FUNCTION auth.get_user_primary_role IS 'Get primary role code for user (for backward compatibility)';

/**
 * Function: auth.get_user_all_roles
 * Returns all currently active roles for a user (respecting temporal bounds)
 */
CREATE OR REPLACE FUNCTION auth.get_user_all_roles(p_user_id BIGINT)
RETURNS TABLE (
  role_id BIGINT,
  role_code VARCHAR(50),
  role_name VARCHAR(100),
  category VARCHAR(50),
  is_primary BOOLEAN,
  effective_from TIMESTAMP,
  effective_to TIMESTAMP
) AS $$
DECLARE
  v_now TIMESTAMP := CURRENT_TIMESTAMP;
BEGIN
  RETURN QUERY
  SELECT 
    hr.id,
    hr.code,
    hr.name,
    hr.category,
    (u.primary_role_id = hr.id),
    ura.effective_from,
    ura.effective_to
  FROM auth.users u
  LEFT JOIN auth.user_role_assignments ura ON ura.user_id = u.id AND ura.status = 'Active'
  LEFT JOIN system.hospital_roles hr ON hr.id = ura.role_id
  WHERE u.id = p_user_id
    AND u.status = 'Active'
    AND (ura.effective_from IS NULL OR ura.effective_from <= v_now)
    AND (ura.effective_to IS NULL OR ura.effective_to >= v_now)
    AND hr.status = 'Active';
END;
$$ LANGUAGE plpgsql STABLE;

COMMENT ON FUNCTION auth.get_user_all_roles IS 'Get all currently active roles for user (temporal filtering applied)';

/**

 * Business logic for assigning a role to a user
 * 
 * Handles:
 *   - Validation that role exists
 *   - Validation that assigned_by user is authorized to assign this role
 *   - Creation of audit trail
 *   - Temporal bounds
 */
CREATE OR REPLACE FUNCTION auth.assign_role_to_user(
  p_user_id BIGINT,
  p_role_code VARCHAR(50),
  p_assigned_by BIGINT,
  p_reason TEXT,
  p_effective_from TIMESTAMP DEFAULT NULL,
  p_effective_to TIMESTAMP DEFAULT NULL
)
RETURNS TABLE (
  success BOOLEAN,
  message VARCHAR(500),
  assignment_id BIGINT
) AS $$
DECLARE
  v_role_id BIGINT;
  v_assignment_id BIGINT;
  v_assigner_role VARCHAR(50);
  v_assignment_authorized BOOLEAN := FALSE;
BEGIN
  -- Validate role exists and is active
  SELECT id INTO v_role_id
  FROM system.hospital_roles
  WHERE code = p_role_code AND status = 'Active';
  
  IF v_role_id IS NULL THEN
    RETURN QUERY SELECT FALSE, 'Role does not exist or is inactive'::VARCHAR(500), NULL::BIGINT;
    RETURN;
  END IF;
  
  -- Validate assigner has permission to assign this role
  SELECT hr.code INTO v_assigner_role
  FROM auth.users u
  LEFT JOIN system.hospital_roles hr ON hr.id = u.primary_role_id
  WHERE u.id = p_assigned_by;
  
  -- TODO: Implement role assignment authorization
  -- For now, only SYSTEM_ADMIN can assign roles
  IF v_assigner_role != 'SYSTEM_ADMIN' THEN
    RETURN QUERY SELECT FALSE, 'User is not authorized to assign roles'::VARCHAR(500), NULL::BIGINT;
    RETURN;
  END IF;
  
  -- Create assignment
  INSERT INTO auth.user_role_assignments (
    user_id, role_id, assigned_by, reason, effective_from, effective_to, status
  ) VALUES (
    p_user_id, v_role_id, p_assigned_by, p_reason, p_effective_from, p_effective_to, 'Active'
  )
  RETURNING id INTO v_assignment_id;
  
  -- TODO: Write audit log
  
  RETURN QUERY SELECT TRUE, 'Role assigned successfully'::VARCHAR(500), v_assignment_id;
END;
$$ LANGUAGE plpgsql;
